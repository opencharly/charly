**Setup launcher — the fixed generated content (pasted):**

```
$ ./bin/charly marketplace generate --root . --out /tmp/out && cat /tmp/out/setup
#!/usr/bin/env bash
# GENERATED by charly marketplace generate — DO NOT EDIT.
# Regenerate this marketplace from the charly checkout its .gitmodules pins:
#   ./setup            generate — cd ./charly && charly marketplace generate --root . --out ..
#   ./setup --check   fail-closed drift gate (regeneration must be a no-op)
set -euo pipefail
cd "$(dirname "$0")/charly"
# The charly CLI prescans the 'marketplace' word from the project root it runs in,
# so the command executes FROM the charly checkout; --root/--out stay explicit.
for arg in "$@"; do
    if [ "$arg" = "--check" ] || [ "$arg" = "--dry-run" ]; then
        exec charly marketplace drift --root . --out ..
    fi
done
exec charly marketplace generate --root . --out ..
```
